const MarqueeText = React.memo(({ text, className = "" }: { text: string; className?: string }) => {
  const containerRef = useRef<HTMLDivElement>(null);
  const textRef = useRef<HTMLSpanElement>(null);
  const [dist, setDist] = useState(0);
  useEffect(() => {
    const measure = () => {

      const d = t.scrollWidth - c.clientWidth;
      setDist(d > 4 ? Math.ceil(d) : 0);

    <div ref={containerRef} className={`overflow-hidden whitespace-nowrap w-full block ${dist ? "mask-edges" : ""} ${className}`}>

const hueFromName = (name: string) => {
  let hash = 0;
  for (let i = 0; i < name.length; i++) hash = name.charCodeAt(i) + ((hash << 5) - hash);
  return Math.abs(hash) % 360;
};
const extractHueFromImage = (src: string): Promise<number> => new Promise((resolve) => {
  const img = new Image();
  img.crossOrigin = "anonymous";
  img.onload = () => {
    try {
      const c = document.createElement("canvas"); c.width = 32; c.height = 32;
      const ctx = c.getContext("2d", { willReadFrequently: true });
      if (!ctx) return resolve(0);
      ctx.drawImage(img, 0, 0, 32, 32);
      const d = ctx.getImageData(0, 0, 32, 32).data;
      let r = 0, g = 0, b = 0, n = 0;
      for (let i = 0; i < d.length; i += 16) {
        if (d[i + 3] < 128) continue;
        r += d[i]; g += d[i + 1]; b += d[i + 2]; n++;
      }
      if (!n) return resolve(0);
      r /= n * 255; g /= n * 255; b /= n * 255;
      const mx = Math.max(r, g, b), mn = Math.min(r, g, b);
      let h = 0;
      if (mx !== mn) {
        const df = mx - mn;
        if (mx === r) h = ((g - b) / df + (g < b ? 6 : 0)) / 6;
        else if (mx === g) h = ((b - r) / df + 2) / 6;
        else h = ((r - g) / df + 4) / 6;
      }
      resolve(Math.round(h * 360));
    } catch { resolve(0); }
  };
  img.onerror = () => resolve(0);
  img.src = src;
});

/* ===================== HELPERS ===================== */
function parseLrc(lrcText: string): LyricLine[] {
  if (!lrcText) return [];
  const lines = lrcText.split(/\r?\n/);
  const result: LyricLine[] = [];
  const timeRegex = /\[(\d{1,3}):(\d{2})(?:\.(\d{1,3}))?\]/g;
  for (const line of lines) {
    const matches = [...line.matchAll(timeRegex)];
    if (matches.length > 0) {
      const text = line.replace(timeRegex, "").trim();
      for (const match of matches) {
        const min = parseInt(match[1], 10);
        const sec = parseInt(match[2], 10);
        const msStr = match[3] || "0";
        const ms = msStr.length === 1 ? parseInt(msStr, 10) * 100 : msStr.length === 2 ? parseInt(msStr, 10) * 10 : parseInt(msStr, 10);
        if (text) result.push({ time: min * 60 + sec + ms / 1000, text });
      }
    }
  }
  return result.sort((a, b) => a.time - b.time);
}
const formatTime = (time: number) => {
  if (!time || isNaN(time) || time < 0) return "0:00";
  const m = Math.floor(time / 60), s = Math.floor(time % 60);
  return `${m}:${s < 10 ? "0" : ""}${s}`;
};
const formatLongTime = (secs: number) => {
  if (!secs || isNaN(secs) || secs <= 0) return "—";
  const h = Math.floor(secs / 3600), m = Math.floor((secs % 3600) / 60);
  if (h > 0) return `${h}h ${m}m`;
  if (m > 0) return `${m} min`;
  return `${Math.floor(secs)}s`;
};
const formatBytes = (bytes: number) => {
  if (!bytes || bytes <= 0) return "0 B";
  const mb = bytes / (1024 * 1024);
  return mb >= 1 ? `${mb.toFixed(1)} MB` : `${(bytes / 1024).toFixed(0)} KB`;
};
const findActiveIdx = (lines: LyricLine[], t: number) => {
  let lo = 0, hi = lines.length - 1, ans = -1;
  while (lo <= hi) {
    const m = (lo + hi) >> 1;
    if (lines[m].time <= t) { ans = m; lo = m + 1; } else hi = m - 1;
  }
  return ans;
};
const customSmoothScroll = (container: HTMLElement, targetPos: number, duration: number) => {
  if ((container as any)._raf) cancelAnimationFrame((container as any)._raf);
  const startPos = container.scrollTop;
  const distance = targetPos - startPos;
  let startTime: number | null = null;
  const ease = (t: number) => t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2;
  const step = (now: number) => {
    if (startTime === null) startTime = now;
    const el = now - startTime, p = Math.min(el / duration, 1);
    container.scrollTop = startPos + distance * ease(p);
    if (el < duration) (container as any)._raf = requestAnimationFrame(step);
  };
  (container as any)._raf = requestAnimationFrame(step);
};

/* ===================== WORD SYNC ===================== */
type WordGroup = { els: HTMLElement[]; s: number[]; e: number[] };
const buildWordGroup = (container: HTMLElement | null, enabled: boolean): WordGroup | null => {
  if (!container || !enabled) return null;
  const els = Array.from(container.querySelectorAll('.lyric-word-sync')) as HTMLElement[];
  if (!els.length) return null;
  let total = 0;
  const lens = els.map(w => { const l = (w.textContent || '').length; total += l; return l; });
  if (!total) return null;
  const s: number[] = [], e: number[] = [];
  let acc = 0;
  lens.forEach(l => { s.push((acc / total) * 100); acc += l; e.push((acc / total) * 100); });
  els.forEach(w => { (w as any)._p = undefined; });
  return { els, s, e };
};

          else v = ((prog - s[i]) / Math.max(0.0001, e[i] - s[i])) * 120;
          if (v === 120 || v === -15) { if (el._p !== v) { el._p = v; el.style.setProperty('--p', `${v}%`); } }

        }
      }
    };
    if (isPlaying) {
      const loop = () => { apply(); raf = requestAnimationFrame(loop); };
      raf = requestAnimationFrame(loop);
    } else apply();
    return () => { if (raf) cancelAnimationFrame(raf); };

                      className="relative z-10 p-2 rounded-xl text-zinc-400 hover:text-rose-400 hover:bg-rose-500/10 transition ml-2"

                    >
                      <Trash2 size={16} />
                    </button>
                  </div>
                );
              })
            )}
          </main>
        </div>
      )}

      {/* ============ MINI PILL ============ */}
      {currentTrack && !isExpanded && (
        <div onClick={() => setIsExpanded(true)}
          className="fixed bottom-[30px] left-3 right-3 h-[60px] rounded-2xl z-[99990] cursor-pointer overflow-hidden transition-all duration-300 shadow-[0_8px_30px_rgba(0,0,0,0.5)] no-select-text border border-white/10"
          style={{ background: `linear-gradient(100deg, ${palette.base} 0%, ${palette.deep} 100%)` }}>
          <div className="absolute inset-0 bg-black/20 pointer-events-none" />
          <div className="relative z-10 w-full h-full flex items-center px-2.5">
            <div className="w-[44px] h-[44px] flex-shrink-0 rounded-[8px] shadow-sm overflow-hidden mr-3">{renderArt(currentTrack, "w-[44px] h-[44px]", "rounded-[8px]")}</div>
            <div className="flex flex-col flex-1 min-w-0 pr-3 justify-center">
              <MarqueeText text={currentTrack.title || currentTrack.baseName} className="text-[13px] font-bold text-white leading-tight mb-[2px] w-full" />
              <MarqueeText text={currentTrack.artist || "Unknown Artist"} className="text-[12px] font-medium text-white/70 leading-tight w-full" />
            </div>
            <div className="flex items-center gap-3 flex-shrink-0 pr-1 text-white">
              <button className="active:scale-75 transition-transform flex items-center justify-center w-[24px] h-[24px]" onClick={(e) => { e.stopPropagation(); playPrev(e); }} aria-label="Previous"><IcPrev size={24} /></button>
              <button className="active:scale-75 transition-transform flex items-center justify-center w-[28px] h-[28px] relative" onClick={togglePlayPause} aria-label={isPlaying ? "Pause" : "Play"}>
                {isBuffering && <div className="spotify-spinner-mini pointer-events-none z-0" />}
                {isPlaying ? <IcPause size={28} /> : <IcPlay size={28} className="translate-x-[1px]" />}
              </button>
              <button className="active:scale-75 transition-transform flex items-center justify-center w-[24px] h-[24px]" onClick={(e) => { e.stopPropagation(); playNext(e); }} aria-label="Next"><IcNext size={24} /></button>
            </div>
          </div>
          <div className="absolute bottom-0 left-2 right-2 h-[2px] bg-white/10 rounded-full z-20 pointer-events-none overflow-hidden">
            <div className="absolute top-0 left-0 h-full bg-white/40 transition-all duration-300" style={{ width: `${buffered}%` }} />
            <div className="absolute top-0 left-0 h-full bg-white rounded-full transition-all duration-300 ease-linear" style={{ width: `${progress}%` }} />
          </div>
        </div>
      )}

      {/* ============ EXPANDED PLAYER ============ */}
      <div className={`player-root fixed inset-0 z-[99999] text-white transition-all duration-[450ms] ease-[cubic-bezier(0.32,0.72,0,1)] ${isExpanded ? "translate-y-0 opacity-100 overflow-hidden" : "translate-y-full opacity-0 pointer-events-none"}`}>
        <div className="absolute inset-0 z-0 pointer-events-none transition-all duration-700"
          style={{ backgroundColor: palette.base, backgroundImage: isLyricsFullScreen ? 'none' : `radial-gradient(120% 55% at 50% 0%, ${palette.glow} 0%, transparent 65%), linear-gradient(to bottom, rgba(0,0,0,0.1) 0%, rgba(0,0,0,0.7) 100%)` }} />

        <div className={`absolute inset-0 z-20 overflow-x-hidden scrollbar-hide flex flex-col pointer-events-none ${isLyricsFullScreen ? 'overflow-y-hidden' : 'overflow-y-auto'}`}>
          <div className="w-full flex flex-col flex-shrink-0 pointer-events-auto transition-all duration-500" style={{ height: isLyricsFullScreen ? '100%' : undefined, minHeight: isLyricsFullScreen ? '100%' : '100dvh' }}>

            {/* Top bar */}
            <div className={`flex items-center justify-between px-5 pt-[max(1rem,env(safe-area-inset-top))] pb-2 flex-shrink-0 w-full ${isLyricsFullScreen ? 'mt-0' : 'mt-4'}`}>
              <button onClick={() => { if (isLyricsFullScreen) setIsLyricsFullScreen(false); else setIsExpanded(false); }} className="p-2 -ml-2 text-white active:opacity-50 drop-shadow-md pointer-events-auto" aria-label="Close"><IcChevronDown size={28} /></button>
              <div className="flex flex-col items-center flex-1 min-w-0 px-2 drop-shadow-md no-select-text">
                <span className="text-[10px] tracking-widest text-white/70 uppercase truncate w-full text-center font-medium">Playing from Downloads</span>
                <span className="text-[13px] font-bold text-white truncate w-full text-center mt-[2px]">{isNativeApk ? "Local Storage" : "Browser Cache"}</span>
              </div>
              <button onClick={(e) => { e.stopPropagation(); setShowSettingsMenu(true); }} className="p-2 -mr-2 text-white active:opacity-50 drop-shadow-md pointer-events-auto" aria-label="Settings"><IcMore size={24} /></button>
            </div>

            {/* Center area */}
            <div className={`flex-1 min-h-0 w-full flex items-center justify-center relative z-30 transition-all duration-500 ${isLyricsFullScreen ? 'px-0 py-0 flex-col items-stretch justify-start' : (isVideo ? 'px-4 py-2' : 'px-8 py-2')}`}>
              {isLyricsFullScreen && isLyricsEnabled ? (
                <div className="flex-1 w-full h-full flex flex-col relative overflow-hidden pointer-events-auto bg-transparent">
                  <div className="flex-1 overflow-y-auto scrollbar-hide px-6 pt-4 pb-[30vh] flex flex-col gap-8 w-full h-full mask-edges-vertical" ref={fullLyricsContainerRef}>
                    {parsedLyrics.length === 0 && <p className="text-white/60 text-center mt-8">No lyrics available for this track.</p>}
                    {RenderedLyrics}
                  </div>
                </div>
              ) : isVideo && currentTrack ? (
                <div className="w-full aspect-video max-w-[600px] max-h-[50vh] relative bg-black shadow-[0_15px_40px_rgba(0,0,0,0.5)] rounded-[12px] overflow-hidden mx-auto pointer-events-auto">
                  {currentTrack && (
                    <video
                      ref={videoRef}
                      src={currentTrack.url}
                      className="w-full h-full object-contain"
                      playsInline
                      controls
                      onTimeUpdate={(e) => {
                        const el = e.currentTarget;
                        if (!isSeekingRef.current) { setCurrentTime(el.currentTime); if (el.duration > 0) setProgress((el.currentTime / el.duration) * 100); }
                        if (el.buffered.length && el.duration > 0) setBuffered((el.buffered.end(el.buffered.length - 1) / el.duration) * 100);
                      }}
                      onLoadedMetadata={(e) => setDuration(e.currentTarget.duration || 0)}
                      onPlay={() => setIsPlaying(true)}
                      onPause={() => setIsPlaying(false)}
                      onEnded={() => playNext()}
                    />
                  )}
                </div>
              ) : currentTrack ? (
                <div className="relative bg-[#282828] rounded-[8px] shadow-[0_15px_40px_rgba(0,0,0,0.5)] overflow-hidden transition-all duration-700 ease-[cubic-bezier(0.32,0.72,0,1)]"
                  style={{ width: '100%', aspectRatio: '1/1', maxWidth: '380px', maxHeight: '50vh', boxShadow: `0 25px 60px -10px ${palette.glow}, 0 15px 40px rgba(0,0,0,0.5)` }}>
                  {renderArt(currentTrack, "w-full h-full", "rounded-[8px]")}
                </div>
              ) : null}
            </div>

            {/* Bottom controls */}
            <div className={`w-full px-5 pt-2 flex flex-col justify-end flex-shrink-0 transition-all duration-500 pointer-events-auto ${isLyricsFullScreen ? 'mb-0 pb-[max(0.5rem,env(safe-area-inset-bottom))]' : 'mb-2 pb-[max(1.5rem,env(safe-area-inset-bottom))]'}`}>

              {currentTrack?.hasLyrics && parsedLyrics.length > 0 && (
                <div className={`transition-all duration-500 w-full relative overflow-hidden flex items-center justify-start mask-edges-vertical ${isUiHidden ? 'max-h-0 opacity-0 mb-0' : (isLyricsFullScreen ? 'hidden' : 'mb-3 opacity-100 min-h-[75px]')}`}>
                  {RenderedMiniLyrics}
                </div>
              )}

              <div className={`transition-all duration-500 flex items-center justify-between drop-shadow-md w-full no-select-text ${isLyricsFullScreen ? 'mb-2 scale-[0.8] origin-bottom-left' : 'mb-5'}`}>
                <div className="flex items-center gap-3 overflow-hidden pr-4 flex-1 min-w-0">
                  {currentTrack && <div className="w-[48px] h-[48px] rounded-md shadow-md flex-shrink-0 overflow-hidden">{renderArt(currentTrack, "w-[48px] h-[48px]", "rounded-md")}</div>}
                  <div className="flex flex-col flex-1 min-w-0">
                    <MarqueeText text={currentTrack?.title || currentTrack?.baseName || "Unknown"} className="text-[22px] font-bold text-white tracking-tight drop-shadow-md w-full" />
                    <MarqueeText text={currentTrack?.artist || "Unknown Artist"} className="text-[15px] font-medium text-[#b3b3b3] mt-1 drop-shadow-md w-full" />
                  </div>
                </div>
              </div>

              <div className={`w-full flex flex-col gap-1 relative drop-shadow-md ${isLyricsFullScreen ? 'mb-2 scale-[0.95] origin-bottom' : 'mb-5'}`}>
                <input type="range" min="0" max="100" value={durationSafe > 0 ? progress : 0}
                  onChange={handleSeekChange}
                  onPointerDown={handleSeekStart} onPointerUp={handleSeekEnd}
                  onTouchStart={handleSeekStart} onTouchEnd={handleSeekEnd}
                  className="w-full mobile-slider relative z-10 pointer-events-auto"
                  style={{ background: `linear-gradient(to right, #fff ${progress}%, rgba(255,255,255,0.3) ${progress}%, rgba(255,255,255,0.3) ${buffered}%, rgba(255,255,255,0.1) ${buffered}%)` }} />
                <div className="flex items-center justify-between text-[11px] font-medium text-[#a7a7a7] mt-1 w-full pointer-events-none no-select-text"><span>{formatTime(currentTime)}</span><span>{formatTime(duration)}</span></div>
              </div>

              <div className={`flex flex-col w-full transition-all duration-[450ms] ease-[cubic-bezier(0.32,0.72,0,1)] ${isUiHidden ? 'max-h-0 opacity-0 translate-y-6 pointer-events-none overflow-hidden' : (isLyricsFullScreen ? 'max-h-[130px] opacity-100 translate-y-0 pointer-events-auto scale-[0.85] origin-bottom overflow-visible' : 'max-h-[160px] opacity-100 translate-y-0 pointer-events-auto overflow-visible')}`}>
                <div className={`flex items-center justify-between w-full px-1 py-1.5 drop-shadow-md no-select-text ${isLyricsFullScreen ? 'mb-0' : 'mb-4'}`}>
                  <button onClick={() => setIsShuffle(!isShuffle)} className={`active:opacity-50 pointer-events-auto ${isShuffle ? 'text-[#1db954]' : 'text-white'}`} aria-label="Shuffle"><IcShuffle size={24} /></button>
                  <button onClick={playPrev} className="text-white active:opacity-50 pointer-events-auto" aria-label="Previous"><IcPrev size={38} /></button>
                  <div className="relative flex items-center justify-center">
                    {isBuffering && <div className="spotify-spinner pointer-events-none z-0" />}
                    <button onClick={togglePlayPause} aria-label={isPlaying ? "Pause" : "Play"}
                      className="w-[64px] h-[64px] rounded-full bg-white flex items-center justify-center text-black active:scale-95 transition-transform shadow-lg relative z-10 pointer-events-auto">
                      {isPlaying ? <IcPause size={28} /> : <IcPlay size={30} className="translate-x-[1px]" />}
                    </button>
                  </div>
                  <button onClick={playNext} className="text-white active:opacity-50 pointer-events-auto" aria-label="Next"><IcNext size={38} /></button>
                  <button onClick={() => setRepeatMode(p => (p + 1) % 3)} className={`active:opacity-50 relative pointer-events-auto ${repeatMode > 0 ? 'text-[#1db954]' : 'text-white/70'}`} aria-label="Repeat">
                    <IcRepeat size={24} />{repeatMode === 2 && <span className="absolute -top-1 -right-1 bg-[#1db954] text-black text-[9px] font-bold rounded-full w-3 h-3 flex items-center justify-center">1</span>}
                  </button>
                </div>
                {!isLyricsFullScreen && (
                  <div className="flex items-center justify-between text-[#b3b3b3] w-full px-1 drop-shadow-md pointer-events-auto mt-2">
                    <div className="flex items-center gap-4">
                      <button onClick={() => { setShowTimerMenu(true); }} className={`active:opacity-50 transition-colors ${sleepTimer ? 'text-[#1db954]' : 'text-[#b3b3b3]'}`} aria-label="Sleep timer"><Timer size={22} /></button>
                    </div>
                    <div className="flex items-center gap-6">
                      <button onClick={() => setShowQueue(true)} className="active:opacity-50 text-white" aria-label="Queue"><IcQueue size={22} /></button>
                    </div>
                  </div>
                )}
              </div>
            </div>
          </div>

          {/* Scrollable secondary area */}
          <div className={`w-full px-5 pb-24 flex flex-col gap-6 pointer-events-auto transition-opacity duration-500 ${isUiHidden ? 'opacity-0 pointer-events-none' : 'opacity-100'} ${isLyricsFullScreen ? 'hidden' : 'block'}`}>
            {isLyricsEnabled && currentTrack?.hasLyrics && parsedLyrics.length > 0 && !isLyricsFullScreen && (
              <div className="rounded-3xl p-6 w-full mx-auto shadow-2xl relative overflow-hidden transition-colors duration-500 border border-white/10"
                style={{ background: `linear-gradient(165deg, ${palette.cards.lyrics[0]} 0%, ${palette.cards.lyrics[1]} 100%)` }}>
                <div className="absolute -top-16 -right-16 w-52 h-52 rounded-full blur-[70px] pointer-events-none" style={{ backgroundColor: palette.glow }} />
                <div className="relative z-10 flex items-center justify-between mb-6 no-select-text">
                  <h3 className="text-white font-bold text-[18px]">Lyrics</h3>
                  <button onClick={() => setIsLyricsFullScreen(true)} className="p-2 text-white/80 hover:text-white rounded-full bg-black/30 pointer-events-auto" aria-label="Fullscreen lyrics"><IcExpand size={16} /></button>
                </div>
                <div className="relative z-10 flex flex-col gap-5 max-h-[300px] overflow-y-auto scrollbar-hide pb-10" ref={lyricsContainerRef}>{RenderedLyrics}</div>
              </div>
            )}

            {currentTrack && (
              <div className="w-full mt-1 rounded-3xl p-5 flex flex-col gap-5 border border-white/10 shadow-[0_20px_50px_rgba(0,0,0,0.45)] relative overflow-hidden no-select-text"
                style={{ background: `linear-gradient(160deg, ${palette.cards.about[0]} 0%, ${palette.cards.about[1]} 100%)` }}>
                <div className="absolute -top-24 -right-20 w-72 h-72 rounded-full blur-[90px] pointer-events-none" style={{ backgroundColor: palette.glow }} />
                <div className="relative z-10 flex items-center gap-3">
                  <div className="w-[52px] h-[52px] rounded-xl overflow-hidden shadow-lg border border-white/10">{renderArt(currentTrack, "w-full h-full", "rounded-xl")}</div>
                  <div className="flex flex-col min-w-0">
                    <h3 className="text-white font-extrabold text-[18px] leading-tight drop-shadow-md flex items-center gap-2"><Sparkles size={17} style={{ color: accent }} /> About this track</h3>
                    <span className="text-white/55 text-[12px] font-medium truncate">{currentTrack.title || currentTrack.baseName}</span>
                  </div>
                </div>
                <div className="relative z-10 grid grid-cols-2 gap-3">
                  <StatTile icon={<Music size={17} />} label="Format" value={currentTrack.ext.toUpperCase()} accent={accent} />
                  <StatTile icon={<Timer size={17} />} label="Size" value={formatBytes(currentTrack.size)} accent="#fcd462" />
                  <StatTile icon={<Type size={17} />} label="Lyrics" value={currentTrack.hasLyrics ? "Synced" : "None"} accent="#f9a8d4" />
                  <StatTile icon={<IcQueue size={17} />} label="Type" value={isVideo ? "Video" : "Audio"} accent="#8cc2ff" />
                </div>
                {currentTrack.artist && (
                  <div className="relative z-10 text-white/70 text-[13px] font-medium">
                    <span className="text-white/50 text-[11px] font-bold uppercase tracking-wider block mb-1">Artist</span>
                    {currentTrack.artist}
                  </div>
                )}
                {currentTrack.album && (
                  <div className="relative z-10 text-white/70 text-[13px] font-medium">
                    <span className="text-white/50 text-[11px] font-bold uppercase tracking-wider block mb-1">Album</span>
                    {currentTrack.album}
                  </div>
                )}
              </div>
            )}
          </div>
        </div>
      </div>

      {/* ============ SETTINGS SHEET ============ */}
      <div className={`fixed inset-0 z-[100000] bg-black/65 transition-opacity duration-300 pointer-events-auto flex flex-col justify-end ${showSettingsMenu ? 'opacity-100' : 'opacity-0 pointer-events-none'}`} onClick={() => setShowSettingsMenu(false)}>
        <div className={`w-full rounded-t-[32px] transition-transform duration-[400ms] ease-[cubic-bezier(0.32,0.72,0,1)] shadow-[0_-20px_60px_rgba(0,0,0,0.6)] border-t border-white/10 flex flex-col max-h-[90vh] overflow-hidden relative ${showSettingsMenu ? 'translate-y-0' : 'translate-y-full'}`}
          style={{ background: `linear-gradient(180deg, ${palette.cards.queue[0]} 0%, #0c0c0e 42%)` }} onClick={e => e.stopPropagation()}>
          <div className="absolute -top-24 left-1/2 -translate-x-1/2 w-[120%] h-56 pointer-events-none opacity-60" style={{ background: `radial-gradient(closest-side, ${palette.glow}, transparent)` }} />
          <div className="relative z-10 flex flex-col items-center pt-3 pb-1 flex-shrink-0"><div className="w-10 h-[5px] rounded-full bg-white/25" /></div>
          <div className="relative z-10 flex items-center justify-between px-6 pt-3 pb-4 flex-shrink-0">
            <h3 className="text-white font-black text-[24px] tracking-tight flex items-center gap-2.5"><Settings2 size={24} style={{ color: accent }} /> Settings</h3>
            <button onClick={() => setShowSettingsMenu(false)} className="text-white/70 p-2 hover:text-white bg-white/10 rounded-full active:scale-90 transition-transform" aria-label="Close"><IcChevronDown size={20} /></button>
          </div>
          <div className="relative z-10 px-5 pb-[max(2.5rem,env(safe-area-inset-bottom))] flex flex-col gap-5 overflow-y-auto scrollbar-hide flex-1">
            <div className="flex items-center gap-3 p-3 rounded-3xl border border-white/10" style={{ background: `linear-gradient(120deg, ${palette.cards.lyrics[0]}, rgba(255,255,255,0.04))` }}>
              <div className="w-14 h-14 rounded-2xl overflow-hidden bg-[#282828] flex-shrink-0 shadow-lg">{currentTrack && renderArt(currentTrack, "w-14 h-14", "rounded-2xl")}</div>
              <div className="flex flex-col min-w-0 flex-1">
                <span className="text-white/55 text-[10px] font-bold uppercase tracking-widest">Now playing · {palette.name.slice(0, 20)}</span>
                <span className="text-white font-extrabold text-[16px] truncate">{currentTrack?.title || currentTrack?.baseName || "—"}</span>
                <span className="text-white/60 text-[12.5px] font-medium truncate">{currentTrack?.artist || "Unknown Artist"}</span>
              </div>
              <EqBars playing={isPlaying} color={accent} />
            </div>

            {/* Lyrics & Word Sync */}
            <div className="flex flex-col gap-2.5">
              <span className="text-white/50 text-[11px] font-bold uppercase tracking-wider pl-2 flex items-center gap-1.5"><Type size={13} /> Lyrics &amp; word sync</span>
              <div className="flex flex-col rounded-3xl bg-white/[0.05] border border-white/[0.07] overflow-hidden">
                <SettingRow accent={accent} icon={<Type size={18} />} title="Show lyrics" sub="Display lyrics on player"
                  right={<Toggle accent={accent} on={isLyricsEnabled} onClick={() => { const n = !isLyricsEnabled; setIsLyricsEnabled(n); localStorage.setItem('off_lyrics_enabled', n.toString()); if (!n) setIsLyricsFullScreen(false); }} />} />

                {isLyricsEnabled && (
                  <div className="flex flex-col gap-4 px-4 py-4 border-b border-white/[0.06] bg-black/20">
                    <div className="flex flex-col gap-2">
                      <span className="text-white/60 text-[12.5px] font-semibold">Mini lyrics size</span>
                      <Segmented accent={accent} value={lineFontSize} options={SIZE_OPTIONS} onChange={(sz) => { setLineFontSize(sz); localStorage.setItem('off_line_fs', sz); }} />
                    </div>
                    <div className="flex flex-col gap-2">
                      <span className="text-white/60 text-[12.5px] font-semibold">Card &amp; fullscreen size</span>
                      <Segmented accent={accent} value={cardFontSize} options={SIZE_OPTIONS} onChange={(sz) => { setCardFontSize(sz); localStorage.setItem('off_card_fs', sz); }} />
                    </div>
                  </div>
                )}

                <SettingRow accent={accent} icon={<Sparkles size={18} />} title="Word sync (card)" sub="Syncs fullscreen & card lyrics"
                  right={<Toggle accent={accent} on={isWordSyncEnabled} onClick={() => { const n = !isWordSyncEnabled; setIsWordSyncEnabled(n); localStorage.setItem('off_word_sync', n.toString()); }} />} />
                <SettingRow last accent={accent} icon={<Sparkles size={18} />} title="Word sync (mini)" sub="Syncs the lyrics above the title"
                  right={<Toggle accent={accent} on={isMiniWordSyncEnabled} onClick={() => { const n = !isMiniWordSyncEnabled; setIsMiniWordSyncEnabled(n); localStorage.setItem('off_mini_word_sync', n.toString()); }} />} />
              </div>
            </div>

            {/* MANAGE — Delete current track */}
            {currentTrack && window.AndroidApp?.deleteDownloadedFile && (
              <div className="flex flex-col gap-2.5">
                <span className="text-white/50 text-[11px] font-bold uppercase tracking-wider pl-2 flex items-center gap-1.5">
                  <Trash2 size={13} /> Manage
                </span>
                <button
                  onClick={() => {
                    const t = currentTrack;
                    if (!t) return;
                    if (!confirm(`Delete "${t.title || t.baseName}"?`)) return;
                    const ok = window.AndroidApp?.deleteDownloadedFile?.(t.name);
                    if (ok) {
                      mediaEl()?.pause();
                      setCurrentTrack(null);
                      setIsPlaying(false);
                      setIsExpanded(false);
                      setShowSettingsMenu(false);
                      loadFiles();
                    }
                  }}
                  className="flex items-center justify-between gap-3 px-4 py-3.5 rounded-2xl bg-rose-500/10 border border-rose-500/25 text-rose-300 active:scale-[0.98] transition"
                >
                  <div className="flex items-center gap-3 min-w-0">
                    <div className="w-9 h-9 rounded-xl flex items-center justify-center bg-rose-500/20 text-rose-300 flex-shrink-0">
                      <Trash2 size={18} />
                    </div>
                    <div className="flex flex-col min-w-0">
                      <span className="font-bold text-[15px] leading-tight">Delete current track</span>
                      <span className="text-white/45 text-[11.5px] font-medium mt-0.5 leading-snug truncate max-w-[220px]">
                        {currentTrack.title || currentTrack.baseName}
                      </span>
                    </div>
                  </div>
                  <X size={16} className="opacity-60 flex-shrink-0" />
                </button>
              </div>
            )}
          </div>
        </div>
      </div>

      {/* ============ QUEUE (now above the player) ============ */}
      <div className={`fixed inset-0 z-[100005] bg-[#09090b] transition-transform duration-300 flex flex-col pointer-events-auto overflow-hidden ${showQueue ? 'translate-y-0' : 'translate-y-full'}`}>
        <div className="absolute inset-0 pointer-events-none" style={{ background: `linear-gradient(180deg, ${palette.cards.queue[0]} 0%, ${palette.cards.queue[1]} 46%, #070708 100%)` }} />
        <div className="absolute -top-28 left-1/2 -translate-x-1/2 w-[150%] h-[380px] pointer-events-none opacity-70" style={{ background: `radial-gradient(closest-side, ${palette.glow}, transparent)` }} />

        <div className="relative z-20 flex items-center justify-between px-5 pt-[max(1.5rem,env(safe-area-inset-top))] pb-3 no-select-text">
          <button onClick={() => setShowQueue(false)} className="w-10 h-10 -ml-2 flex items-center justify-center rounded-full bg-white/10 text-white active:scale-90 transition-transform" aria-label="Close queue"><IcChevronDown size={24} /></button>
          <div className="flex flex-col items-center min-w-0 px-3">
            <span className="text-[16px] font-extrabold text-white tracking-tight">Queue</span>
            <span className="text-[10px] font-bold text-white/55 uppercase tracking-widest truncate max-w-[220px]">Downloads</span>
          </div>
          <div className="w-10" />
        </div>

        <div className="relative z-10 flex-1 overflow-y-auto px-5 pb-[220px] no-select-text scrollbar-hide" ref={queueContainerRef}>
          {/* Now playing */}
          {currentTrack && (
            <div className="flex items-center justify-between w-full mb-5 p-3.5 rounded-3xl border border-white/[0.12] relative overflow-hidden shadow-[0_16px_40px_rgba(0,0,0,0.35)]"
              style={{ background: `linear-gradient(120deg, ${palette.cards.lyrics[0]} 0%, rgba(255,255,255,0.05) 100%)` }}>
              <div className="flex items-center gap-3.5 overflow-hidden relative z-10 min-w-0">
                <div className="w-16 h-16 flex-shrink-0 rounded-2xl overflow-hidden shadow-lg">{renderArt(currentTrack, "w-16 h-16", "rounded-2xl")}</div>
                <div className="flex flex-col min-w-0 pr-2 overflow-hidden">
                  <span className="text-[10px] font-extrabold uppercase tracking-widest mb-0.5" style={{ color: accent }}>Now playing</span>
                  <span className="text-[17px] font-extrabold text-white truncate">{currentTrack.title || currentTrack.baseName}</span>
                  <span className="text-[13px] font-medium text-white/65 truncate">{currentTrack.artist || "Unknown Artist"}</span>
                </div>
              </div>
              <div className="relative z-10 pr-1 flex-shrink-0"><EqBars playing={isPlaying} color={accent} /></div>
              <div className="absolute bottom-0 left-0 h-[3px] w-full bg-white/10"><div className="h-full" style={{ width: `${progress}%`, background: `linear-gradient(to right, ${accent}, ${palette.light})` }} /></div>
            </div>
          )}

          {/* Stats */}
          {filteredTracks.length > 0 && (
            <div className="grid grid-cols-3 gap-2.5 mb-6">
              {[
                { v: String(filteredTracks.length), l: "Tracks" },
                { v: String(filteredTracks.filter(t => !t.isVideo).length), l: "Audio" },
                { v: String(filteredTracks.filter(t => t.isVideo).length), l: "Video" },
              ].map((s, i) => (
                <div key={i} className="flex flex-col gap-0.5 p-3 rounded-2xl border border-white/[0.08]" style={{ background: `linear-gradient(145deg, ${palette.cards.stats[0]}, ${palette.cards.stats[1]})` }}>
                  <span className="text-white font-extrabold text-[18px] tracking-tight leading-tight truncate">{s.v}</span>
                  <span className="text-white/55 text-[11px] font-semibold">{s.l}</span>
                </div>
              ))}
            </div>
          )}

          <div className="flex items-center justify-between mb-3">
            <span className="text-[17px] font-extrabold text-white">On this device</span>
            <span className="text-[12px] font-semibold text-white/45">{filteredTracks.length} songs</span>
          </div>
          {filteredTracks.length === 0 && <p className="text-white/45 text-[14px] font-medium py-8 text-center">Nothing queued.</p>}
          <div className="flex flex-col relative">
            {filteredTracks.map((track, i) => {
              const isCurrent = currentTrack?.name === track.name;
              return (
                <div key={track.name + i}
                  onClick={() => { playTrack(track, true); setShowQueue(false); }}
                  className={`queue-item flex items-center justify-between w-full group h-[60px] p-2 rounded-xl cursor-pointer relative ${isCurrent ? 'bg-white/[0.1]' : 'bg-transparent hover:bg-white/[0.06]'}`}>
                  <div className="flex items-center gap-3 overflow-hidden flex-1 min-w-0">
                    <div className="w-[44px] h-[44px] flex-shrink-0 rounded-[8px] overflow-hidden shadow-md">{renderArt(track, "w-[44px] h-[44px]", "rounded-[8px]")}</div>
                    <div className="flex flex-col min-w-0 pr-2 overflow-hidden">
                      <span className={`text-[15px] font-bold truncate ${isCurrent ? 'text-emerald-300' : 'text-white'}`}>{track.title || track.baseName}</span>
                      <span className="text-[13px] font-medium text-white/60 truncate">{track.artist || "Unknown Artist"}</span>
                    </div>
                  </div>
                  <div className="flex items-center flex-shrink-0 pr-2">
                    {isCurrent ? <EqBars playing={isPlaying} color={accent} /> : <span className="text-[11px] font-semibold text-white/35 uppercase">{track.ext}</span>}
                  </div>
                </div>
              );
            })}
          </div>
        </div>

        {/* Floating toolbar — 35px above bottom */}
        <div className="absolute bottom-[35px] left-0 w-full z-20 px-5 pb-0 pt-3 no-select-text pointer-events-none">
          <div className="pointer-events-auto rounded-[28px] border border-white/[0.12] bg-black/55 backdrop-blur-2xl px-5 py-3 flex justify-between items-center shadow-[0_-10px_40px_rgba(0,0,0,0.5)]">
            <div className="flex flex-col items-center gap-1 active:opacity-50 cursor-pointer min-w-[56px]" onClick={() => setIsShuffle(!isShuffle)}>
              <IcShuffle size={22} className={isShuffle ? '' : 'text-white/70'} />
              <span className="text-[11px] font-semibold" style={{ color: isShuffle ? accent : 'rgba(255,255,255,0.7)' }}>Shuffle</span>
            </div>
            <div className="flex flex-col items-center gap-1 active:opacity-50 cursor-pointer min-w-[56px]" onClick={() => setRepeatMode(p => (p + 1) % 3)}>
              <div className="relative" style={{ color: repeatMode > 0 ? accent : 'rgba(255,255,255,0.7)' }}>
                <IcRepeat size={22} />{repeatMode === 2 && <span className="absolute -top-1 -right-1 text-black text-[9px] font-bold rounded-full w-3 h-3 flex items-center justify-center" style={{ background: accent }}>1</span>}
              </div>
              <span className="text-[11px] font-semibold" style={{ color: repeatMode > 0 ? accent : 'rgba(255,255,255,0.7)' }}>Repeat</span>
            </div>
            <div className="flex flex-col items-center gap-1 active:opacity-50 cursor-pointer min-w-[56px]" onClick={() => setShowTimerMenu(true)}>
              <div style={{ color: sleepTimer ? accent : 'rgba(255,255,255,0.7)' }}><Timer size={22} /></div>
              <span className="text-[11px] font-semibold" style={{ color: sleepTimer ? accent : 'rgba(255,255,255,0.7)' }}>
                {timerRemaining ? formatTime(timerRemaining) : sleepTimer === 'end' ? 'End' : 'Timer'}
              </span>
            </div>
          </div>
        </div>
      </div>

      {/* ============ SLEEP TIMER ============ */}
      {showTimerMenu && (
        <div className="fixed inset-0 z-[100010] bg-black/60 flex items-center justify-center p-6 backdrop-blur-sm" onClick={() => setShowTimerMenu(false)}>
          <div className="w-full max-w-sm rounded-3xl p-6 shadow-2xl flex flex-col gap-2 border border-white/10"
            style={{ background: `linear-gradient(180deg, ${palette.cards.queue[0]} 0%, #151517 70%)` }} onClick={e => e.stopPropagation()}>
            <h4 className="text-white font-bold text-lg mb-2 flex justify-between items-center">Sleep Timer <button onClick={() => setShowTimerMenu(false)} className="text-white/50 hover:text-white"><X size={20} /></button></h4>
            {[5, 15, 30, 45, 60].map(mins => (
              <button key={mins} onClick={() => { setSleepTimer(mins); setShowTimerMenu(false); }}
                className={`py-3 px-4 rounded-xl flex justify-between items-center transition-colors ${sleepTimer === mins ? 'bg-[#1db954]/20 text-[#1db954]' : 'bg-white/5 text-white hover:bg-white/10'}`}>
                <span className="font-medium">{mins} minutes</span>{sleepTimer === mins && <Check size={18} />}
              </button>
            ))}
            <button onClick={() => { setSleepTimer('end'); setShowTimerMenu(false); }}
              className={`py-3 px-4 rounded-xl flex justify-between items-center transition-colors ${sleepTimer === 'end' ? 'bg-[#1db954]/20 text-[#1db954]' : 'bg-white/5 text-white hover:bg-white/10'}`}>
              <span className="font-medium">End of track</span>{sleepTimer === 'end' && <Check size={18} />}
            </button>
            <button onClick={() => { setSleepTimer(null); setShowTimerMenu(false); }} className="py-3 px-4 rounded-xl text-white/50 hover:bg-white/5 text-left mt-2 border border-white/10 transition-colors">Turn off timer</button>
          </div>
        </div>
      )}
    </>
  );
}
